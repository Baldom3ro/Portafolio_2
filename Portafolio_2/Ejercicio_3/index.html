💻 Implementación Service Worker (showNotification) 

// 1. Solicitud de permiso en app.js
const permission = await Notification.requestPermission();

// 2. Registro y obtención del Service Worker
const registration = await navigator.serviceWorker.ready;

// 3. Disparo de notificación local enriquecida
registration.showNotification(title, {
  body: 'Texto descriptivo...',
  icon: './icon-192.png',
  badge: './icon.svg',
  vibrate: [200, 100, 200],
  tag: 'pwa-alerta',
  renotify: true,
  actions: [
    { action: 'explore', title: 'Ver Detalles' },
    { action: 'dismiss', title: 'Cerrar' }
  ]
});

// 4. En sw.js se atiende el clic:
self.addEventListener('notificationclick', (event) => {
  event.notification.close();
  if (event.action === 'explore') {
    event.waitUntil(clients.openWindow('./'));
  }
});
